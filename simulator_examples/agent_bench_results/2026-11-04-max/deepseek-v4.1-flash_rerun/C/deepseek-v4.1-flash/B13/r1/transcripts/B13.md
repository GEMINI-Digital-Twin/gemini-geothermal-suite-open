        """Heat rejected to air; zero when the water is not warmer than the air."""
        p = self.parameters
        c_water = p["water_density"] * p["water_cp"] * u["water_flow_rate"]

            "water_temperature_out": t_out,
            "fan_power": p["fan_power_fraction"] * q,

        }
